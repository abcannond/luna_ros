Commands to run and what they do: 

git clone git@github.com:abcannond/luna_ros.git

cd luna_ros

git submodule update --init --recursive


docker build -t luna_ros:latest .   (build docker container, period is necesarry)

docker build --no-cache -t luna_ros:latest .

./run_ros_image.sh luna_ros:latest

cd ros2_ws

# install/setup.bash is created by colcon build; source it only AFTER building
source /opt/ros/jazzy/setup.bash
colcon build --symlink-install

source install/setup.bash

# ============================================================
# TERMINAL 1: GAZEBO SIMULATION
# ============================================================
ros2 launch lunabot_2425 gz_bringup.launch.py

# Wait for Gazebo to fully start before proceeding to Terminal 2!

# ============================================================
# TERMINAL 2: RTAB-MAP + NAV2 COSTMAP PIPELINE
# ============================================================
# Open NEW terminal and exec into container:
docker exec -it <CONTAINER_ID> /bin/bash
source /ros2_ws/install/setup.bash

# Launch the RTAB-Map + Nav2 integration:
ros2 launch luna_mapping rtabmap_nav2_sim.launch.py

# This launches:
#   - RTAB-Map SLAM (creates occupancy grid from depth camera)
#   - Depth-to-LaserScan (for Nav2 obstacle detection)
#   - Depth-to-PointCloud (for 3D voxel costmap)
#   - Nav2 navigation stack (controller, planner, behaviors)
#
# Launch arguments:
#   use_sim_time:=true          (default: true)
#   use_rtabmap_odom:=false     (use Gazebo odom, set true for visual odom)
#   use_fiducial_odom:=false    (enable fiducial marker localization)
#   launch_nav2:=true           (default: true)
#   launch_rviz:=false          (set true if not using gz_bringup's rviz)
#
# Example with visual odometry:
#   ros2 launch luna_mapping rtabmap_nav2_sim.launch.py use_rtabmap_odom:=true
#
# Topics used:
#   - /camera/camera/color/image_raw        (RGB from D455)
#   - /camera/camera/depth/image_rect_raw   (Depth from D455)
#   - /camera/camera/depth/color/points     (PointCloud2 aligned to color)
#   - /scan                                 (LaserScan from depth)
#   - /map                                  (Occupancy grid from RTAB-Map)
#   - /odom                                 (Odometry)
#   - /cmd_vel                              (Velocity commands from Nav2)
#   - /local_costmap/costmap                (Nav2 local costmap)
#   - /global_costmap/costmap               (Nav2 global costmap)
#   - /plan                                 (Global path)
#   - /local_plan                           (Local path)
#
# Send navigation goals using RViz 2D Nav Goal tool
# ============================================================

# ============================================================
# TERMINAL 3: MONITORING/DIAGNOSTICS (Optional)
# ============================================================
docker exec -it <CONTAINER_ID> /bin/bash
source /ros2_ws/install/setup.bash

# Check topics
ros2 topic list
ros2 topic hz /map
ros2 topic hz /scan
ros2 topic hz /camera/camera/depth/color/points

# Check TF tree
ros2 run tf2_tools view_frames
ros2 run tf2_ros tf2_echo odom base_link

# ============================================================
# TERMINAL 4: RViz VISUALIZATION (Optional)
# ============================================================
docker exec -it <CONTAINER_ID> /bin/bash
source /ros2_ws/install/setup.bash

# Launch RViz with RTAB-Map config
rviz2 -d $(ros2 pkg prefix luna_mapping)/share/luna_mapping/config/rtabmap_nav2.rviz

# Or launch manually:
rviz2
# Then add displays: Map, Costmaps, PointCloud2, LaserScan, RobotModel, TF

# ============================================================
# TERMINAL 5: TELEOP CONTROL (Optional)
# ============================================================
docker exec -it <CONTAINER_ID> /bin/bash
source /ros2_ws/install/setup.bash

# Use the drive alias (sends to /cmd_vel)
drive

# Or manually:
ros2 run teleop_twist_keyboard teleop_twist_keyboard --ros-args -r cmd_vel:=/cmd_vel

# Controls: i=forward, ,=backward, j=left, l=right, k=stop
